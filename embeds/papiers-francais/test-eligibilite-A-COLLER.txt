      titre: 'Bonne nouvelle : vous réunissez les conditions essentielles',
      message: "D'après vos réponses, rien ne bloque votre démarche « " + LABEL_DEMARCHE[d] + " ». Il reste à consolider votre dossier et à préparer l'examen civique"
        + (d === 'naturalisation' ? " et l'entretien d'assimilation" : '')
        + ' pour mettre toutes les chances de votre côté. Papiers Français peut sécuriser le montage de votre dossier et vous mettre en relation avec un avocat partenaire.',
      checklist: checklist,
      etapes: etapesBase,
      routage: fiscalOk ? 'formation' : 'mixte',
      qualification: fiscalOk
        ? 'CHAUD — conditions réunies : prospect prioritaire'
        : 'Conditions réunies mais point fiscal à régler'
    };
  }

  // ═══ ÉTAT ══════════════════════════════════════════════════════════════
  var racine = document.getElementById('pf-eligibilite');
  var phase = 'intro';
  var reponses = {};
  var index = 0;
  var lead = { prenom: '', nom: '', email: '', telephone: '', rgpd: false };
  var envoi = false;
  var resultat = null;

  function esc(s) {
    return String(s).replace(/[&<>"']/g, function (c) {
      return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c];
    });
  }

  // ═══ ENVOI DU LEAD ═════════════════════════════════════════════════════
  function envoyer() {
    var v = verdict(reponses);
    resultat = v;

    var charge = {
      source: 'test-eligibilite-papiers-francais',
      recu_le: new Date().toISOString(),
      prenom: lead.prenom.trim(),
      nom: lead.nom.trim(),
      email: lead.email.trim().toLowerCase(),
      telephone: lead.telephone.trim(),
      consentement: lead.rgpd === true,
      consentement_le: lead.rgpd ? new Date().toISOString() : null,
      demarche: LABEL_DEMARCHE[reponses.demarche] || reponses.demarche || '',
      verdict: v.titre,
      statut: v.status,
      routage: v.routage,
      qualification: v.qualification,
      urgent: v.status === 'regularisation',
      reponses: reponses,
      page: location.href,
      referer: document.referrer || null,
      utm_source: new URLSearchParams(location.search).get('utm_source'),
      utm_medium: new URLSearchParams(location.search).get('utm_medium'),
      utm_campaign: new URLSearchParams(location.search).get('utm_campaign')
    };

    // L'envoi ne doit jamais empêcher l'affichage du résultat : une panne
    // réseau ferait perdre le lead ET la réponse promise à la personne.
    if (PF_WEBHOOK) {
      try {
        fetch(PF_WEBHOOK, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(charge)
        })['catch'](function () {});
      } catch (e) { /* ignoré volontairement */ }
    } else if (window.console) {
      console.info('[PF éligibilité] PF_WEBHOOK non renseigné — lead non transmis :', charge);
    }

    if (window.dataLayer) {
      window.dataLayer.push({ event: 'pf_lead_eligibilite', statut: v.status, routage: v.routage });
    }
  }

  // ═══ RENDU ═════════════════════════════════════════════════════════════
  function fleche() {
    return '<svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.5" stroke-linecap="round" stroke-linejoin="round"><path d="M5 12h14M12 5l7 7-7 7"/></svg>';
  }

  function rendre() {
    var visibles = questionsVisibles(reponses);

    if (phase === 'intro') {
      racine.innerHTML =
        '<div class="pfel-centre">' +
          '<span class="pfel-pastille">' +
            '<svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.4" stroke-linecap="round" stroke-linejoin="round"><path d="M12 22s8-4 8-10V5l-8-3-8 3v7c0 6 8 10 8 10z"/></svg>' +
            'Test gratuit · 2 min</span>' +
          '<h1 class="pfel-h1">Testez votre éligibilité</h1>' +
          '<p class="pfel-intro">Carte de séjour, carte de résident ou naturalisation : répondez à quelques questions pour savoir où vous en êtes et ce qu’il vous reste à préparer.</p>' +
          '<button class="pfel-btn" data-act="start">Commencer le test ' + fleche() + '</button>' +
          '<p class="pfel-note">Outil informatif — ne remplace pas une décision de la préfecture ni un avis juridique.</p>' +
        '</div>';
      return;
    }

    if (phase === 'quiz') {
      var q = visibles[index];
      if (!q) { phase = 'contact'; return rendre(); }
      var pct = Math.round((index / visibles.length) * 100);
      var txt = aide(q, reponses);

      racine.innerHTML =
        '<div class="pfel-barre"><div class="pfel-barre-fill" style="width:' + pct + '%"></div></div>' +
        '<div class="pfel-etape"><span>Question ' + (index + 1) + ' / ' + visibles.length + '</span><span>' + pct + ' %</span></div>' +
        '<div class="pfel-carte">' +
          '<p class="pfel-theme">' + esc(q.theme) + '</p>' +
          '<h2 class="pfel-h2">' + esc(q.question) + '</h2>' +
          (txt ? '<p class="pfel-aide">' + esc(txt) + '</p>' : '') +
          q.options.map(function (o) {
            return '<button class="pfel-opt" data-act="rep" data-val="' + esc(o.value) + '">' + esc(o.label) + '</button>';
          }).join('') +
          (index > 0 ? '<button class="pfel-lien" data-act="retour">← Question précédente</button>' : '') +
        '</div>';
      return;
    }

    if (phase === 'contact') {
      var pret = lead.prenom && lead.nom && lead.email && lead.telephone && lead.rgpd;
      racine.innerHTML =
        '<div class="pfel-barre"><div class="pfel-barre-fill" style="width:100%"></div></div>' +
        '<div class="pfel-etape"><span>Dernière étape</span><span>100 %</span></div>' +
        '<div class="pfel-carte">' +
          '<h2 class="pfel-h2">Votre résultat est prêt</h2>' +
          '<p class="pfel-aide">Indiquez vos coordonnées pour l’afficher et être recontacté·e par un conseiller Papiers Français.</p>' +
          '<div class="pfel-duo">' +
            champ('prenom', 'Prénom', 'Marie', 'text') +
            champ('nom', 'Nom', 'Dupont', 'text') +
          '</div>' +
          champ('email', 'Email', 'marie.dupont@email.com', 'email') +
          champ('telephone', 'Téléphone', '06 12 34 56 78', 'tel') +
          '<label class="pfel-rgpd">' +
            '<input type="checkbox" data-act="rgpd"' + (lead.rgpd ? ' checked' : '') + '>' +
            '<span>J’accepte d’être recontacté·e par <strong>Papiers Français</strong> au sujet de mes démarches, et que mes coordonnées soient transmises, si ma situation le justifie, à un <strong>avocat partenaire</strong> spécialisé en droit des étrangers. Je suis informé·e que mon adresse IP est enregistrée afin de vérifier la zone d\u2019intervention. Données traitées conformément au RGPD ; je peux retirer mon consentement à tout moment.</span>' +
          '</label>' +
          '<button class="pfel-btn pfel-btn-plein" data-act="voir"' + (pret && !envoi ? '' : ' disabled') + '>' +
            (envoi ? 'Analyse en cours…' : 'Voir mon résultat ' + fleche()) +
          '</button>' +
          '<button class="pfel-lien" data-act="retour-quiz">← Revenir aux questions</button>' +
        '</div>';
      return;
    }

    // ── Résultat ──
    var v = resultat;
    var puces = { true: ['#0E7A58', '✓'], warn: ['#B45309', '!'], false: ['#C0332B', '×'] };

    racine.innerHTML =
      (v.urgent
        ? '<div class="pfel-urgence"><strong>Délais de recours très courts.</strong> Contre une OQTF, le délai est souvent de 48 heures à 30 jours selon la mesure notifiée. Chaque jour compte.</div>'
        : '') +
      '<div class="pfel-bandeau pfel-b-' + v.ton + '">' +
        '<h2>' + esc(v.titre) + '</h2>' +
        '<p>' + esc(v.message) + '</p>' +
      '</div>' +
      (v.checklist.length
        ? '<div class="pfel-carte" style="margin-bottom:18px">' +
            '<p class="pfel-bloc-titre">Votre situation point par point</p>' +
            v.checklist.map(function (c) {
              var p = puces[String(c.ok)];
              return '<div class="pfel-check">' +
                '<span class="pfel-puce" style="background:' + p[0] + '">' + p[1] + '</span>' +
                '<span>' + esc(c.label) + '</span></div>';
            }).join('') +
          '</div>'
        : '') +
      '<div class="pfel-carte">' +
        '<p class="pfel-bloc-titre">Ce qu’il vous reste à faire</p>' +
        '<ol class="pfel-etapes">' +
          v.etapes.map(function (e) { return '<li>' + esc(e) + '</li>'; }).join('') +
        '</ol>' +
      '</div>' +
      '<div class="pfel-cta">' +
        '<h2 class="pfel-h2" style="font-size:18px">Un conseiller vous recontacte sous 48 h</h2>' +
        '<p class="pfel-aide" style="margin:8px 0 16px">Votre demande est enregistrée. Pour aller plus vite, vous pouvez aussi prendre rendez-vous dès maintenant.</p>' +
        '<a class="pfel-btn' + (v.urgent ? ' pfel-btn-rouge' : '') + '" href="' + PF_RDV_URL + '" target="_blank" rel="noopener">' +
          (v.urgent ? 'Être rappelé en urgence ' : 'Prendre rendez-vous ') + fleche() +
        '</a>' +
      '</div>' +
      '<p class="pfel-note pfel-centre">Résultat indicatif, fondé sur vos déclarations. Il ne constitue ni une décision de l’administration ni un avis juridique.</p>';
  }

  function champ(id, label, ph, type) {
    return '<div class="pfel-champ">' +
      '<label class="pfel-label" for="pfel-' + id + '">' + label + '</label>' +
      '<input class="pfel-input" id="pfel-' + id + '" type="' + type + '" data-champ="' + id + '" ' +
      'value="' + esc(lead[id]) + '" placeholder="' + esc(ph) + '" autocomplete="' +
      ({ prenom: 'given-name', nom: 'family-name', email: 'email', telephone: 'tel' })[id] + '">' +
    '</div>';
  }

  // ═══ ÉVÉNEMENTS ════════════════════════════════════════════════════════
  racine.addEventListener('click', function (ev) {
    var el = ev.target.closest('[data-act]');
    if (!el) return;
    var act = el.getAttribute('data-act');

    if (act === 'start') { phase = 'quiz'; index = 0; return rendre(); }

    if (act === 'rep') {
      var visibles = questionsVisibles(reponses);
      var q = visibles[index];
      reponses[q.id] = el.getAttribute('data-val');
      // La liste des questions dépend des réponses : on la recalcule avant
      // d'avancer, sinon on saute ou on répète une question conditionnelle.
      var apres = questionsVisibles(reponses);
      if (index + 1 >= apres.length) { phase = 'contact'; } else { index = index + 1; }
      return rendre();
    }

    if (act === 'retour') { index = Math.max(0, index - 1); return rendre(); }

    if (act === 'retour-quiz') {
      phase = 'quiz';
      index = Math.max(0, questionsVisibles(reponses).length - 1);
      return rendre();
    }

    if (act === 'voir') {
      if (envoi) return;
      if (!(lead.prenom && lead.nom && lead.email && lead.telephone && lead.rgpd)) return;
      envoi = true;
      rendre();
      envoyer();
      setTimeout(function () { phase = 'resultat'; envoi = false; rendre(); window.scrollTo({ top: racine.offsetTop - 40, behavior: 'smooth' }); }, 650);
      return;
    }
  });

  // On met à jour l'état sans re-rendre : un re-rendu à chaque frappe ferait
  // perdre le focus et la position du curseur.
  racine.addEventListener('input', function (ev) {
    var c = ev.target.getAttribute && ev.target.getAttribute('data-champ');
    if (c) {
      lead[c] = ev.target.value;
      majBouton();
    }
  });

  racine.addEventListener('change', function (ev) {
    if (ev.target.getAttribute && ev.target.getAttribute('data-act') === 'rgpd') {
      lead.rgpd = ev.target.checked;
      majBouton();
    }
  });

  function majBouton() {
    var b = racine.querySelector('[data-act="voir"]');
    if (!b) return;
    var pret = lead.prenom && lead.nom && lead.email && lead.telephone && lead.rgpd;
    if (pret) b.removeAttribute('disabled'); else b.setAttribute('disabled', '');
  }

  rendre();
})();
</script>
